# วิธีแก้แบบช้า

วิธีตรงไปตรงมาคือคำนวณผลรวมของอาร์เรย์ย่อยทุกชุดที่เป็นไปได้

เริ่มจากสมาชิกแต่ละตัว แล้วหาผลรวมของอาร์เรย์ย่อยทุกชุดที่เริ่มจากสมาชิกตัวนั้น

เช่น สำหรับ `[-1, 2, 3, -9, 11]`:

```js no-beautify
// เริ่มจาก -1:
-1
-1 + 2
-1 + 2 + 3
-1 + 2 + 3 + (-9)
-1 + 2 + 3 + (-9) + 11

// เริ่มจาก 2:
2
2 + 3
2 + 3 + (-9)
2 + 3 + (-9) + 11

// เริ่มจาก 3:
3
3 + (-9)
3 + (-9) + 11

// เริ่มจาก -9
-9
-9 + 11

// เริ่มจาก 11
11
```

เมื่อลงมือเขียนจะได้ลูปซ้อนกัน ลูปนอกเลือกจุดเริ่มต้น ส่วนลูปในคำนวณผลรวมของอาร์เรย์ย่อยที่เริ่มจากจุดนั้น

```js run
function getMaxSubSum(arr) {
  let maxSum = 0; // ถ้าไม่เลือกสมาชิกตัวใดเลย จะคืนค่าเป็นศูนย์

  for (let i = 0; i < arr.length; i++) {
    let sumFixedStart = 0;
    for (let j = i; j < arr.length; j++) {
      sumFixedStart += arr[j];
      maxSum = Math.max(maxSum, sumFixedStart);
    }
  }

  return maxSum;
}

alert( getMaxSubSum([-1, 2, 3, -9]) ); // 5
alert( getMaxSubSum([-1, 2, 3, -9, 11]) ); // 11
alert( getMaxSubSum([-2, -1, 1, 2]) ); // 3
alert( getMaxSubSum([1, 2, 3]) ); // 6
alert( getMaxSubSum([100, -9, 2, -3, 5]) ); // 100
```

วิธีนี้มีความซับซ้อนด้านเวลาเป็น [O(n<sup>2</sup>)](https://en.wikipedia.org/wiki/Big_O_notation) ถ้าอาร์เรย์ยาวขึ้น 2 เท่า อัลกอริทึมจะใช้เวลาประมาณ 4 เท่า

สำหรับอาร์เรย์ขนาดใหญ่ เช่น มีสมาชิก 1,000, 10,000 ตัว หรือมากกว่านั้น วิธีแบบนี้อาจทำให้โปรแกรมทำงานช้ามาก

# วิธีแก้แบบเร็ว

วนอาร์เรย์หนึ่งรอบและเก็บผลรวมย่อยปัจจุบันไว้ในตัวแปร `partialSum` ถ้าเมื่อไร `partialSum` ติดลบ ให้ตั้งค่าเป็น `0` แล้วเริ่มสะสมใหม่

ค่าสูงสุดที่ `partialSum` เคยมีคือคำตอบ

ถ้ายังนึกภาพไม่ออก ลองไล่จากโค้ดสั้น ๆ นี้:

```js run demo
function getMaxSubSum(arr) {
  let maxSum = 0;
  let partialSum = 0;

  for (let item of arr) { // วนผ่านสมาชิกแต่ละตัวของ arr
    partialSum += item; // บวกค่านี้เข้า partialSum
    maxSum = Math.max(maxSum, partialSum); // เก็บค่าสูงสุดไว้
    if (partialSum < 0) partialSum = 0; // ถ้าติดลบให้กลับเป็นศูนย์
  }

  return maxSum;
}

alert( getMaxSubSum([-1, 2, 3, -9]) ); // 5
alert( getMaxSubSum([-1, 2, 3, -9, 11]) ); // 11
alert( getMaxSubSum([-2, -1, 1, 2]) ); // 3
alert( getMaxSubSum([100, -9, 2, -3, 5]) ); // 100
alert( getMaxSubSum([1, 2, 3]) ); // 6
alert( getMaxSubSum([-1, -2, -3]) ); // 0
```

อัลกอริทึมนี้วนอาร์เรย์เพียงรอบเดียว ความซับซ้อนด้านเวลาจึงเป็น O(n)

รายละเอียดเพิ่มเติมอยู่ใน [ปัญหาอาร์เรย์ย่อยที่มีผลรวมมากที่สุด](http://en.wikipedia.org/wiki/Maximum_subarray_problem)

ถ้ายังไม่เห็นว่าทำไมวิธีนี้ถึงได้ผล ลองไล่ค่าของตัวแปรทีละรอบกับตัวอย่างข้างบน จะเห็นการทำงานชัดกว่าการอ่านคำอธิบายอย่างเดียว
