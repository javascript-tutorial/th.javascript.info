# เลือกสมาชิก (element) ที่ไม่ซ้ำกันในอาร์เรย์ (array)

กำหนดให้ `arr` เป็นอาร์เรย์

เขียนฟังก์ชัน (function) `unique(arr)` ที่คืนค่าอาร์เรย์ของสมาชิกใน `arr` โดยให้แต่ละค่าปรากฏเพียงครั้งเดียว

ตัวอย่าง:

```js
function unique(arr) {
  /* เขียนโค้ดตรงนี้ */
}

let strings = ["Hare", "Krishna", "Hare", "Krishna",
  "Krishna", "Krishna", "Hare", "Hare", ":-O"
];

alert( unique(strings) ); // Hare, Krishna, :-O
```
