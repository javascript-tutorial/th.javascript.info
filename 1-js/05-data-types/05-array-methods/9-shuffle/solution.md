เราอาจลองเขียนแบบนี้ก่อน:

```js run
*!*
function shuffle(array) {
  array.sort(() => Math.random() - 0.5);
}
*/!*

let arr = [1, 2, 3];
shuffle(arr);
alert(arr);
```

โค้ดนี้สลับลำดับสมาชิก (element) ได้ เพราะ `Math.random() - 0.5` ให้ค่าสุ่มที่อาจเป็นบวกหรือลบ ฟังก์ชัน (function) เรียงลำดับจึงเปลี่ยนตำแหน่งสมาชิกไปตามค่าที่สุ่มได้

แต่ฟังก์ชันเรียงลำดับไม่ได้ออกแบบมาให้ใช้งานแบบนี้ การเรียงสับเปลี่ยน (permutation) แต่ละรูปแบบจึงไม่ได้มีโอกาสเกิดเท่ากัน

ลองดูโค้ดนี้ เราเรียก `shuffle` 1,000,000 ครั้ง แล้วนับว่าแต่ละรูปแบบเกิดขึ้นกี่ครั้ง:

```js run
function shuffle(array) {
  array.sort(() => Math.random() - 0.5);
}

// นับจำนวนครั้งที่แต่ละรูปแบบเกิดขึ้น
let count = {
  '123': 0,
  '132': 0,
  '213': 0,
  '231': 0,
  '321': 0,
  '312': 0
};

for (let i = 0; i < 1000000; i++) {
  let array = [1, 2, 3];
  shuffle(array);
  count[array.join('')]++;
}

// แสดงจำนวนครั้งของแต่ละรูปแบบ
for (let key in count) {
  alert(`${key}: ${count[key]}`);
}
```

ตัวอย่างผลลัพธ์ ซึ่งอาจต่างกันตามเอนจิน JavaScript (JavaScript engine) ที่ใช้:

```js
123: 250706
132: 124425
213: 249618
231: 124880
312: 125148
321: 125223
```

ผลลัพธ์ชุดนี้แสดงให้เห็นชัดว่าโอกาสไม่เท่ากัน เพราะ `123` และ `213` เกิดขึ้นบ่อยกว่ารูปแบบอื่นมาก

แม้ผลที่ได้อาจต่างกันในแต่ละเอนจิน JavaScript แต่วิธีนี้ก็ไม่รับประกันว่าทุกรูปแบบจะมีโอกาสเกิดเท่ากัน

สาเหตุอยู่ที่ฟังก์ชันเปรียบเทียบ ปกติเราให้อาร์เรย์ (array) กับฟังก์ชันเปรียบเทียบแก่ `sort` แล้วให้ `sort` จัดลำดับตามผลการเปรียบเทียบ แต่การสุ่มทำให้เปรียบเทียบสมาชิกคู่เดิมแล้วอาจได้ผลต่างกัน ผลลัพธ์จึงขึ้นอยู่กับวิธีเรียงลำดับที่แต่ละเอนจินใช้

วิธีที่เหมาะกับงานนี้คืออัลกอริทึม (algorithm) [Fisher-Yates shuffle](https://en.wikipedia.org/wiki/Fisher%E2%80%93Yates_shuffle) ซึ่งไล่จากท้ายอาร์เรย์มาด้านหน้า ในแต่ละรอบจะสุ่มตำแหน่งตั้งแต่ต้นอาร์เรย์จนถึงตำแหน่งปัจจุบัน แล้วสลับสมาชิกสองตำแหน่งนั้น:

```js
function shuffle(array) {
  for (let i = array.length - 1; i > 0; i--) {
    let j = Math.floor(Math.random() * (i + 1)); // สุ่มตำแหน่งตั้งแต่ 0 ถึง i

    // สลับสมาชิก array[i] กับ array[j]
    // ใช้รูปแบบ destructuring assignment เพื่อสลับค่า
    // เราจะอธิบายรูปแบบนี้ในบทถัด ๆ ไป
    // เขียนอีกแบบให้ได้ผลเดียวกันได้ดังนี้:
    // let t = array[i]; array[i] = array[j]; array[j] = t
    [array[i], array[j]] = [array[j], array[i]];
  }
}
```

ลองทดสอบด้วยวิธีเดิม:

```js run
function shuffle(array) {
  for (let i = array.length - 1; i > 0; i--) {
    let j = Math.floor(Math.random() * (i + 1));
    [array[i], array[j]] = [array[j], array[i]];
  }
}

// นับจำนวนครั้งที่แต่ละรูปแบบเกิดขึ้น
let count = {
  '123': 0,
  '132': 0,
  '213': 0,
  '231': 0,
  '321': 0,
  '312': 0
};

for (let i = 0; i < 1000000; i++) {
  let array = [1, 2, 3];
  shuffle(array);
  count[array.join('')]++;
}

// แสดงจำนวนครั้งของแต่ละรูปแบบ
for (let key in count) {
  alert(`${key}: ${count[key]}`);
}
```

คราวนี้จำนวนครั้งที่แต่ละรูปแบบเกิดขึ้นใกล้เคียงกัน สอดคล้องกับการสุ่มที่ให้โอกาสทุกรูปแบบเท่ากัน

Fisher-Yates ยังทำงานได้มีประสิทธิภาพกว่า เพราะสลับสมาชิกโดยตรง ไม่ต้องเรียกกระบวนการเรียงลำดับ
