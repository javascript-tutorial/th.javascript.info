# แก้การบวกให้ถูกต้อง

โค้ดนี้ให้ผู้ใช้กรอกตัวเลขสองตัว แล้วแสดงผลบวกของตัวเลขทั้งสอง

แต่โค้ดยังทำงานไม่ถูกต้อง เมื่อใช้ค่าเริ่มต้นใน `prompt` ผลลัพธ์จะเป็น `12`

เกิดจากอะไร? ลองแก้ให้ผลลัพธ์เป็น `3`

```js run
let a = prompt("ตัวเลขตัวแรก?", 1);
let b = prompt("ตัวเลขตัวที่สอง?", 2);

alert(a + b); // 12
```
