เมื่อใช้ backtick JavaScript จะคำนวณนิพจน์ใน `${...}` แล้วนำผลลัพธ์มาแทรกในสตริง

```js run
let name = "Ilya";

// นิพจน์คือเลข 1
alert( `สวัสดี ${1}` ); // สวัสดี 1

// นิพจน์คือสตริง "name"
alert( `สวัสดี ${"name"}` ); // สวัสดี name

// นิพจน์คือตัวแปร จึงนำค่าของตัวแปรมาแทรก
alert( `สวัสดี ${name}` ); // สวัสดี Ilya
```
